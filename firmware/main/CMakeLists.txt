idf_component_register(SRCS "main.c" "config_write_guard.c" "input_router.c" "device_services.c" "device_status.c" "binding_table.c" "ota_client.c" "ota_stream.c" "control.c" "serial_input.c" "serial_transport.c" "serial_session.c" INCLUDE_DIRS "." REQUIRES app_update esp_driver_usb_serial_jtag esp_app_format esp_driver_uart esp_driver_gpio esp_timer nvs_flash)

set(XIAO_ANTENNA "" CACHE STRING "XIAO antenna: external or internal")
set_property(CACHE XIAO_ANTENNA PROPERTY STRINGS external internal)
if(XIAO_ANTENNA STREQUAL "external")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE XIAO_EXTERNAL_ANTENNA=1)
elseif(XIAO_ANTENNA STREQUAL "internal")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE XIAO_EXTERNAL_ANTENNA=0)
else()
    message(FATAL_ERROR "XIAO_ANTENNA must be external or internal")
endif()

# Stable releases use major.minor.patch.0; USB-only prerelease probes sort just below it.
string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-.*)?$" ota_semver "${PROJECT_VER}")
if(NOT ota_semver)
    message(FATAL_ERROR "PROJECT_VER must be a semantic firmware version")
endif()
set(ota_major "${CMAKE_MATCH_1}")
set(ota_minor "${CMAKE_MATCH_2}")
set(ota_patch "${CMAKE_MATCH_3}")
set(ota_prerelease "${CMAKE_MATCH_4}")
if(ota_major GREATER 255 OR ota_minor GREATER 255 OR ota_patch GREATER 255)
    message(FATAL_ERROR "OTA version components must fit in one byte")
endif()
math(EXPR ota_file_version "(${ota_major} << 24) | (${ota_minor} << 16) | (${ota_patch} << 8)")
if(ota_prerelease)
    math(EXPR ota_file_version "${ota_file_version} - 1")
endif()
target_compile_definitions(${COMPONENT_LIB} PRIVATE XIAO_OTA_FILE_VERSION=${ota_file_version}U)
